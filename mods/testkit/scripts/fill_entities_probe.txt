# Entities spawned on ground a big /tk fill just changed aren't drawn, though the server has
# them (Tester 2, 11-04). Logs each one as the server and the client have it ("entities:" lines:
# position, block under, light, whether the client knows it at all), with shots looking down
# at them, from above, after leaving and coming back, and a control without the fill.
# Positive pitch looks up: these look down.
loaded
wait 3
chat /tk mode creative
chat /tk time 6000
chat /weather clear
where start
# 1. The fill, then five animals 5-11 blocks to +z (yaw 180 faces +z).
chat /tk fill ~-10 ~-1 ~-2 ~10 ~-1 ~16 base:grass
chat /tk fill ~-10 ~ ~-2 ~10 ~8 ~16 air
wait 1
chat /tk spawn animals:cat ~-4 ~ ~5
chat /tk spawn animals:grey_wolf ~0 ~ ~6
chat /tk spawn animals:jaguar ~4 ~ ~6
chat /tk spawn animals:raptor ~-4 ~ ~10
chat /tk spawn animals:capybara ~4 ~ ~11
wait 2
entities 20
look 180 -25
wait 1
shot fill_a_down.png
wait 10
entities 20
look 180 -25
wait 1
shot fill_b_down.png
# 2. From 12 blocks up, looking down at the pen.
chat /tp ~ ~12 ~
look 180 -60
wait 2
entities 32
shot fill_c_above.png
chat /tp ~ ~-12 ~
# 3. Away (beyond the send distance) and back: the client forgets and spawns them again.
chat /tp ~200 ~ ~
wait 4
chat /tp ~-200 ~ ~
wait 4
entities 20
look 180 -25
wait 1
shot fill_d_back.png
# 4. Control, 60 blocks to +x on untouched ground: the same animals, no fill.
chat /tp ~60 ~ ~
wait 3
where control
chat /tk spawn animals:grey_wolf ~0 ~ ~6
chat /tk spawn animals:capybara ~4 ~ ~8
wait 2
entities 12
look 180 -25
wait 1
shot fill_e_control.png
quit
