# Fletching (parts::fletched): a bronze arrowhead, a stick and a feather in the crafting table's
# assembly slots make 4 bronze arrows, item level 16 (vitals::gear::arrow_level, bronze tier 1
# depth 6). One copy, a fresh world. The table menu's slots: grid 0-8, result 9, assembly 10-13,
# assembled 14, apart 15, the bag 16-42, the hotbar 43-51.
# Passes when chat shows "base:arrow 4" (/tk slot) and "item level 16" (/inspect); the shots show
# the row filled, the arrows taken and the row emptied.
loaded
idle
chat /tk mode survival
until mode survival within 10
chat /tk fill ~-3 ~-1 ~-3 ~3 ~-1 ~3 base:stone
chat /tk fill ~-3 ~ ~-3 ~3 ~3 ~3 core:air
chat /tk set ~ ~1 ~-2 base:crafting_table
until block ~ ~1 ~-2 base:crafting_table within 10
look 0 0
# 1. The three pieces, into hotbar slots 0, 1, 2 (menu slots 43, 44, 45).
chat /gear part arrowhead bronze slot=0
wait 1
chat /tk give base:stick 1
wait 1
chat /tk give base:feather 1
wait 1
chat /tk inv me
until chat base:feather within 5
# 2. Into the assembly row.
press core:use
until screen core:menu within 10
wait 1
tap_once core:menu slot:43
tap_once core:menu slot:10
tap_once core:menu slot:44
tap_once core:menu slot:11
tap_once core:menu slot:45
tap_once core:menu slot:12
wait 1
shot fletching_row.png
# 3. Take the arrows and put them in hotbar slot 0.
tap_once core:menu slot:14
tap_once core:menu slot:43
wait 1
shot fletching_taken.png
key Escape
until screen none within 10
# 4. Four arrows, bronze-headed, item level 16; the head, stick and feather used up.
chat /tk slot me 0
until chat base:arrow 4 within 5
chat /inspect
until chat item level 16 within 5
chat /tk inv me
wait 1
shot fletching_done.png
quit
