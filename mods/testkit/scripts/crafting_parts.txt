# Crafting slice 2, parts: make a stone axe head, an oak haft and a cordage lashing from the book's
# Parts tab, build the axe in the table's assembly slots, then take it apart again. One copy, a
# fresh world. The table menu's slots: grid 0-8, result 9, assembly 10-13, assembled 14, apart 15,
# the bag 16-42, the hotbar 43-51. Each part is the only one its materials make, so it's first.
loaded
idle
chat /tk mode survival
until mode survival within 10
chat /tk fill ~-3 ~-1 ~-3 ~3 ~-1 ~3 base:stone
chat /tk fill ~-3 ~ ~-3 ~3 ~3 ~3 core:air
chat /tk set ~ ~1 ~-2 base:crafting_table
until block ~ ~1 ~-2 base:crafting_table within 10
look 0 0
# 1. The head: 3 cobblestone.
chat /tk give base:cobblestone 3
press core:use
until screen core:menu within 10
wait 1
tap core:menu text:parts
wait 1
shot parts_tab.png
tap_once core:menu class:slot
wait 1
shot parts_head_picked.png
tap core:menu text:all (
wait 1
key Escape
until screen none within 10
# 2. The haft: split from 1 oak log.
chat /tk give base:oak_log 1
press core:use
until screen core:menu within 10
wait 1
tap core:menu text:parts
tap_once core:menu class:slot
wait 1
shot parts_haft_picked.png
tap core:menu text:all (
wait 1
key Escape
until screen none within 10
# 3. The lashing: 1 string.
chat /tk give base:string 1
press core:use
until screen core:menu within 10
wait 1
tap core:menu text:parts
tap_once core:menu class:slot
wait 1
tap core:menu text:all (
wait 1
chat /tk inv me
until chat base:part_lashing within 5
shot parts_made.png
# 4. Build: head, haft and lashing from the hotbar into the assembly slots, take the axe.
tap_once core:menu slot:43
tap_once core:menu slot:10
tap_once core:menu slot:44
tap_once core:menu slot:11
tap_once core:menu slot:45
tap_once core:menu slot:12
wait 1
shot parts_assembly.png
tap_once core:menu slot:14
tap_once core:menu slot:43
wait 1
shot parts_built.png
chat /tk slot me 0
until chat base:axe within 5
key Escape
until screen none within 10
# 5. Apart: the axe into the apart slot; head and haft come back, the lashing is cut.
press core:use
until screen core:menu within 10
wait 1
tap_once core:menu slot:43
tap_once core:menu slot:15
wait 1
shot parts_apart.png
key Escape
until screen none within 10
chat /tk inv me
quit
